% Fixed random seed: tagpdf draws its user-namespace ID from it, so the PDF is
% byte-identical across builds (dates come from SOURCE_DATE_EPOCH, see latexmkrc)
\ExplSyntaxOn \sys_gset_rand_seed:n {1} \ExplSyntaxOff
% Tagged PDF/UA-2: structure and reading order are embedded for parsers and screen readers

% ---------- IBM Plex Sans ----------
% Load Plex from TeX Live (via kpathsea) rather than any system-installed copy,
% so the PDF does not depend on what is in ~/Library/Fonts
\edef\plexpath{\directlua{tex.sprint((kpse.find_file("IBMPlexSans-Regular.otf", "opentype fonts"):gsub("[^/]*$", "")))}}
\defaultfontfeatures{Path = \plexpath, Extension = .otf, Ligatures = TeX}
% Use Plex everywhere, including text the class leaves unstyled (e.g. honor commas)
\setmainfont{IBMPlexSans}[
  UprightFont    = *-Regular,
  ItalicFont     = *-Italic,
  BoldFont       = *-SemiBold,
  BoldItalicFont = *-SemiBoldItalic,
]
\setsansfont{IBMPlexSans}[
  UprightFont    = *-Regular,
  ItalicFont     = *-Italic,
  BoldFont       = *-SemiBold,
  BoldItalicFont = *-SemiBoldItalic,
]
\newfontfamily\plexsansfont{IBMPlexSans}[
  UprightFont    = *-Regular,
  ItalicFont     = *-Italic,
  BoldFont       = *-SemiBold,        % Plex Bold is heavy; SemiBold reads better
  BoldItalicFont = *-SemiBoldItalic,
  % Plex has no small caps: map the class's \scshape to the upright faces
  SmallCapsFont  = *-Regular,
  BoldFeatures   = {SmallCapsFont = *-SemiBold},
]
\newfontfamily\plexsanslightfont{IBMPlexSans}[
  UprightFont    = *-Light,
  ItalicFont     = *-LightItalic,
  BoldFont       = *-Medium,
  BoldItalicFont = *-MediumItalic,
  SmallCapsFont  = *-Light,
  BoldFeatures   = {SmallCapsFont = *-Medium},
]

\renewcommand*{\headerfont}{\plexsansfont}
\renewcommand*{\headerfontlight}{\plexsanslightfont}
\renewcommand*{\footerfont}{\plexsansfont}
\renewcommand*{\bodyfont}{\plexsansfont}
\renewcommand*{\bodyfontlight}{\plexsanslightfont}
% Plex has no small caps, so give the positions a readable size instead
\renewcommand*{\headerpositionstyle}[1]{{\fontsize{12pt}{1em}\bodyfont\color{awesome} #1}}
\renewcommand*{\entrypositionstyle}[1]{{\fontsize{9pt}{1em}\bodyfont\color{graytext} #1}}
% Smaller name, larger contact line
\renewcommand*{\headerfirstnamestyle}[1]{{\fontsize{24pt}{1em}\headerfontlight\color{graytext} #1}}
\renewcommand*{\headerlastnamestyle}[1]{{\fontsize{24pt}{1em}\headerfont\bfseries\color{text} #1}}
\renewcommand*{\headersocialstyle}[1]{{\fontsize{7.8pt}{1em}\headerfont\color{text} #1}}
% Smaller section headings
\renewcommand*{\sectionstyleface}[1]{{\fontsize{13pt}{1em}\bodyfont\bfseries #1}}

% Character protrusion and font expansion for tighter, more even lines
